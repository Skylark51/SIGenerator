cmake_minimum_required(VERSION 3.24)
project(SIGeneratorSingleFile LANGUAGES CXX RC)

if(NOT WIN32 OR NOT DEFINED SI_PAYLOAD_DIRECTORY)
  message(FATAL_ERROR "Windows and SI_PAYLOAD_DIRECTORY are required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
configure_file(payload.rc.in payload.rc @ONLY)
add_executable(si_single_file WIN32 launcher.cpp "${CMAKE_CURRENT_BINARY_DIR}/payload.rc")
target_compile_definitions(si_single_file PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(si_single_file PRIVATE cabinet bcrypt shell32 ole32)
set_target_properties(si_single_file PROPERTIES OUTPUT_NAME "SI generator")
if(MSVC)
  target_compile_options(si_single_file PRIVATE /W4 /permissive- /EHsc /O2 /utf-8 /Gy /Gw)
  target_link_options(si_single_file PRIVATE /OPT:REF /OPT:ICF)
else()
  target_compile_options(si_single_file PRIVATE -Os -ffunction-sections -fdata-sections)
  target_link_options(si_single_file PRIVATE -municode -static -static-libgcc -static-libstdc++ -s -Wl,--gc-sections)
endif()
